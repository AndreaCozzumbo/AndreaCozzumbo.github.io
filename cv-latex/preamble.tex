% One line with a right-aligned date; the text wraps before reaching the date.
\newcommand{\datedline}[2]{%
  {\noindent\begin{minipage}[t]{0.80\textwidth}\raggedright #1\strut\end{minipage}\hfill
   \begin{minipage}[t]{0.19\textwidth}\raggedleft\textit{#2}\strut\end{minipage}\par}\vspace{0.1ex}}
